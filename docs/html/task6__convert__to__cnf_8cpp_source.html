154 Node *distr(Node *node1, Node *node2, int depth = 0)
155 {
156 if (!node1 || !node2)
157 return nullptr;
158

198 Node *cnf(Node *currentNode, int depth = 0)
199 {
200 if (!currentNode)
201 return nullptr;
202
203 const int MAX_DEPTH = 2000;
204 string indent(depth * 2, ' '); // Debug variable, unused in logic
205
206 if (depth > MAX_DEPTH)
207 {
208 return copyNode(currentNode);
209 }

211 Node *result = nullptr;
212
213 switch (currentNode->data)
214 {
215 case '*':
216 {
217 // CNF(A * B) => CNF(A) * CNF(B)
218 Node *leftCNF = cnf(currentNode->left, depth + 1);
219 Node *rightCNF = cnf(currentNode->right, depth + 1);
220 result = conjunction(leftCNF, rightCNF);
221 break;
222 }
223 case '+':
224 {
225 // CNF(A + B) => distr(CNF(A), CNF(B))
226 Node *leftCNF = cnf(currentNode->left, depth + 1);
227 Node *rightCNF = cnf(currentNode->right, depth + 1);
228 result = distr(leftCNF, rightCNF, depth + 1);
229 break;
230 }
231 default:
232 // Literal (atom or ~atom)
233 result = copyNode(currentNode); // Always return a safe copy
234 break;
235 }
236
237 return result;
238 }
