160 bool evaluateCNF(const vector<pair<set<char>, set<char>>> &clause_literals,
161 const map<char, bool> &truth_values)
162 {
163 // Iterate over all clauses (Conjunction)
164 for (const auto &clause : clause_literals)
165 {
166 const set<char> &positive = clause.first;
167 const set<char> &negative = clause.second;
168
169 bool clause_value = false; // Is this clause true?
170
171 // Check all positive literals (Disjunction)
172 for (char p : positive)
173 {
174 auto it = truth_values.find(p);
175 if (it != truth_values.end() && it->second)
176 { // if p is 'T'
177 clause_value = true;
178 break;
179 }
180 }
181
182 // Check all negative literals (Disjunction)
183 if (!clause_value)
184 {
185 for (char n : negative)
186 {
187 auto it = truth_values.find(n);
188 if (it != truth_values.end() && !it->second)
189 { // if n is 'F' (so ~n is 'T')
190 clause_value = true;
191 break;
192 }
193 }
194 }
195
196 // If even one clause is false, the whole CNF is false
197 if (!clause_value)
198 {
199 return false;
200 }
201 }
202 // All clauses were true, so the CNF is true
203 return true;
204 }

217 void assignTruthValues(int bitNumber, map<char, bool> &truth_values,
218 int numOfAtoms)
219 {
220 // Extract keys into a vector to ensure consistent order
221 vector<char> atoms;
222 for (auto &kv : truth_values)
223 {
224 atoms.push_back(kv.first);
225 }
226
227 // Assign truth values based on the bits of bitNumber
228 for (int i = 0; i < numOfAtoms; ++i)
229 {
230 bool bit = (bitNumber >> i) & 1;
231 // Assign bits in reverse to match standard truth table order
232 truth_values[atoms[numOfAtoms - 1 - i]] = bit;
233 }

253void generateTruthTable(Node *cnfRootNode, set<char> &atoms)
254{
255 getAtomsRecursive(cnfRootNode, atoms);
256 int numOfAtoms = atoms.size();

267 collectClausesWithLiterals(cnfRootNode, clause_literals);
268
269 int max = (1 << numOfAtoms) - 1; // e.g., for 3 atoms, max is 7 (111)
270
271 // Print table header
272 cout << string(numOfAtoms * 5 + 3, '-') << endl;
273 for (char atom : atoms)

276 }
277 cout << "OUT" << endl;
278 cout << string(numOfAtoms * 5 + 3, '-') << endl;
279
280 // Iterate through all $2^N$ rows
281 for (int i = max; i >= 0; i--)
282 {
283 assignTruthValues(i, truth_values, numOfAtoms);
284
285 // Evaluate and get the result for this row
