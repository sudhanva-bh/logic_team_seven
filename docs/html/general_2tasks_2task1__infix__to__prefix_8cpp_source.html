40string infixToPrefix(string infix)
41{
42 // Step 1: Reverse the infix expression.
43 reverse(infix.begin(), infix.end());
44
45 // Step 2: Swap opening and closing parentheses.
46 // This is crucial for correctly handling operator precedence
47 // after the string has been reversed.
48 for (size_t i = 0; i < infix.length(); ++i)
49 {
50 if (infix[i] == '(')
51 {
52 infix[i] = ')';
53 }
54 else if (infix[i] == ')')
55 {
56 infix[i] = '(';
57 }
58 }
59
60 // Step 3: Convert the modified infix string to postfix.
61 stack<char> opStack;
62 string postfix;
63
64 for (char token : infix)
65 {
66 // If the token is an operand (letter or number), add it to the output.
67 if (isalnum(token))
68 {
69 postfix += token;
70 }
71 // If it's an opening parenthesis, push it onto the operator stack.
72 else if (token == '(')
73 {
74 opStack.push(token);
75 }
76 // If it's a closing parenthesis, pop operators from the stack
77 // to the output until the matching opening parenthesis is found.
78 else if (token == ')')
79 {
80 while (!opStack.empty() && opStack.top() != '(')
81 {
82 postfix += opStack.top();
83 opStack.pop();
84 }
85 opStack.pop(); // Pop and discard the opening parenthesis '('.
86 }
87 // If the token is an operator...
88 else if (isOperator(token))
89 {
90 // Pop operators from the stack that have higher precedence
91 // than the current token.
92 while (!opStack.empty() && opStack.top() != '(' &&
93 getPrecedence(token) < getPrecedence(opStack.top()))
94 {
95 postfix += opStack.top();
96 opStack.pop();
97 }
98 // Push the current operator onto the stack.
99 opStack.push(token);
100 }
101 }
102
103 // After processing all tokens, pop any remaining operators from the stack.
104 while (!opStack.empty())
105 {
106 postfix += opStack.top();
107 opStack.pop();
108 }
109
110 // Step 4: Reverse the postfix string to get the final prefix.
111 reverse(postfix.begin(), postfix.end());
112
113 return postfix;
